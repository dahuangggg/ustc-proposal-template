% \flowrecord{补齐高度}：写入 .aux 供下一遍使用。补齐高度 = 剩余高度 + 本遍已补的高度，使多遍编译结果稳定；
% 边框后的 1mm 间距在页底会被舍弃，可一并补入；预留 2pt，避免舍入误差使边框溢出到下一页。
% 数值变化时提示需要再编译一遍。
\newcommand{\flowrecord}[1]{%
  \ifnum\value{flowbox}>0
    \setlength{\flowgap}{#1}%
    \ifdim\flowgap<0pt \setlength{\flowgap}{0pt}\fi
    \immediate\write\csname @auxout\endcsname{\string\flowfillset{\theflowbox}{\the\flowgap}}%
    \ifdim\dimexpr\flowgap-\flowextra\relax>.5pt \flowrerun\fi
    \ifdim\dimexpr\flowextra-\flowgap\relax>.5pt \flowrerun\fi
  \fi}
\newcommand{\flowrerun}{\PackageWarningNoLine{kt}{Frame heights changed. Rerun to get cross-references right}}
\newcommand{\flowheading}[1]{\par\penalty10000
  \begingroup\dimen0=\pagegoal\advance\dimen0-\pagetotal
  \ifdim\dimen0<\flowneed \flowrecord{\dimexpr\dimen0+\flowextra+1mm-2pt\relax}\clearpage\else\flowrecord{0pt}\fi\endgroup
  \noindent{\sffamily\fontsize{15}{19}\selectfont #1}\par\nopagebreak\vspace{1mm}}

\makeatletter
% 图表标题按中文习惯排为“图 1　标题”，不用冒号。
\long\def\@makecaption#1#2{\vskip\abovecaptionskip{\centering #1\quad #2\par}\vskip\belowcaptionskip}
\newcommand{\flowfloats}{%
  \renewenvironment{figure}[1][]{\par\addvspace{\medskipamount}\centering\def\@captype{figure}}{\par\addvspace{\medskipamount}}%
  \renewenvironment{table}[1][]{\par\addvspace{\medskipamount}\centering\def\@captype{table}}{\par\addvspace{\medskipamount}}}
\makeatother
% 试排正文以测量高度时会让计数器多走一遍：试排前后保存并恢复，避免图表、脚注编号跳号。
\newcommand{\flowsavecounters}{\edef\flowrestorecounters{%
  \noexpand\setcounter{figure}{\the\value{figure}}\noexpand\setcounter{table}{\the\value{table}}%
  \noexpand\setcounter{footnote}{\the\value{footnote}}\noexpand\setcounter{equation}{\the\value{equation}}}}
% 边框内的脚注会丢失：框内只排脚注编号并暂存脚注内容，边框结束后统一输出到页面底部。
\newtoks\flownotes
\newcommand{\flownotesetup}{\renewcommand{\footnote}[1]{\footnotemark
  \begingroup\edef\x{\endgroup\global\flownotes{\the\flownotes
    \noexpand\footnotetext[\the\value{footnote}]{\unexpanded{##1}}}}\x}}
\newcommand{\flownotesflush}{\the\flownotes\global\flownotes{}}
\newcommand{\flowbox}[2]{%
  \setlength{\FrameRule}{0.3pt}\setlength{\FrameSep}{2mm}\setlength{\topsep}{0pt}\setlength{\partopsep}{0pt}%
  % 以空白模板的留白为最小高度；正文超出时由 framed 自动分页。
  \stepcounter{flowbox}%
  \setlength{\flowextra}{0pt}%
  \ifcsname flowfill@\theflowbox\endcsname
    \setlength{\flowextra}{\csname flowfill@\theflowbox\endcsname}%
  \fi
  \setlength{\flowremaining}{#2}%
  \flowsavecounters
  \sbox{\flowmeasure}{\begin{minipage}{\dimexpr\linewidth-2\FrameSep-2\FrameRule\relax}%
    \flowfloats\renewcommand{\footnote}[1]{\footnotemark}\fontsize{10.5}{15}\selectfont #1\par\end{minipage}}%
  \flowrestorecounters
  \addtolength{\flowremaining}{-\ht\flowmeasure}%
  \addtolength{\flowremaining}{-\dp\flowmeasure}%
  % 当前页放不下整个框、但放得下正文时，留白只填到页底，避免空白边框溢出到下一页。
  % \penalty10000 不允许断页，仅促使 TeX 先把标题计入当前页，使 \pagetotal 为最新值。
  \par\penalty10000
  \ifdim\pagegoal<\maxdimen
    \setlength{\flowroom}{\dimexpr\pagegoal-\pagetotal-1mm-2\FrameSep-2\FrameRule-12pt\relax}%
    \ifdim\dimexpr\ht\flowmeasure+\dp\flowmeasure+\flowremaining\relax>\flowroom
      \ifdim\dimexpr\ht\flowmeasure+\dp\flowmeasure\relax<\flowroom
        \setlength{\flowremaining}{\dimexpr\flowroom-\ht\flowmeasure-\dp\flowmeasure\relax}%
      \fi
    \fi
  \fi
  % 补齐到页底的高度加在留白之后（正文已超过最小高度时，从正文末尾起算）。
  \ifdim\flowextra>0pt
    \ifdim\flowremaining<0pt \setlength{\flowremaining}{0pt}\fi
    \addtolength{\flowremaining}{\flowextra}%
  \fi
  % 跨页时不在页底留下段落首行、也不把段落末行单独带到下一页。
  \begin{framed}\flowfloats\flownotesetup\clubpenalty=10000 \widowpenalty=10000 \fontsize{10.5}{15}\selectfont #1\par
  \ifdim\flowremaining>0pt\vspace*{\flowremaining}\fi
  \end{framed}\flownotesflush\vspace{1mm}}
